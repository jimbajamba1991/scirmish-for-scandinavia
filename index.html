// Hamburger menu logic const hamburger = document.getElementById('hamburger-menu'); const headerBtns = document.getElementById('header-btns'); hamburger.addEventListener('click', function() { headerBtns.classList.toggle('open'); }); hamburger.addEventListener('keydown', function(e) { if (e.key === 'Enter' || e.key === ' ') { headerBtns.classList.toggle('open'); } });
